  <script>
    // 1. FIREBASE CONFIGURATION

    firebase.initializeApp(firebaseConfig);
    const auth = firebase.auth();
    const db = firebase.database();

    // 2. CPAGRIP LOCKER LINK
    const CPAGRIP_LOCKER_URL = "https://www.cpagrip.com/show.php?l=YOUR_LOCKER_LINK_HERE";

    // 3. APP STATE
    let currentUser = null;
    let userCoins = 0;
    let currentQuizIndex = 0;
    let roundCoins = 0;
    let quizTimer = null;
    let secondsLeft = 15;

    // KBC Questions Pool (General Knowledge & Entertainment)
    const kbcQuestions = [
      { q: "Bharat ke Rashtrapati ka aadhikarik niwas kahan hai?", o: ["Lal Qila", "Rashtrapati Bhavan", "Sansad Bhavan", "India Gate"], a: 1 },
      { q: "Surya ke sabse kareeb grah (planet) konsa hai?", o: ["Budh (Mercury)", "Shukra (Venus)", "Mangal (Mars)", "Brihaspati"], a: 0 },
      { q: "Cricket game me kitne khiladi har team me hote hain?", o: ["9", "10", "11", "12"], a: 2 },
      { q: "Mobile operating system 'Android' kis company ka product hai?", o: ["Apple", "Microsoft", "Google", "Samsung"], a: 2 },

    ];

    // SPLASH SCREEN INITIALIZATION (2.5 Seconds Progress)
    window.addEventListener('DOMContentLoaded', () => {
      let progress = 0;
      const bar = document.getElementById('splashProgressBar');
      const status = document.getElementById('splashStatus');

      const interval = setInterval(() => {
        progress += 10;
        bar.style.width = progress + '%';

        if (progress === 50) status.innerText = "Connecting secure wallet...";
        if (progress === 80) status.innerText = "Loading KBC challenge...";

        if (progress >= 100) {
          clearInterval(interval);
          checkAuthPersistence();
        }
      }, 150);
    });

    // FIREBASE AUTH PERSISTENCE
    function checkAuthPersistence() {
      auth.onAuthStateChanged((user) => {
        const splash = document.getElementById('splashScreen');
        if (user) {
          currentUser = user;
          document.getElementById('userDisplayName').innerText = user.displayName || "User";
          loadUserData(user.uid);
          splash.classList.add('hidden');
          showScreen('tabHome');
        } else {
          splash.classList.add('hidden');
          document.getElementById('loginScreen').classList.remove('hidden');
        }
      });
    }

    // GOOGLE LOGIN HANDLER
    function handleGoogleSignIn() {
      const provider = new firebase.auth.GoogleAuthProvider();
      auth.signInWithPopup(provider)
        .then((result) => {
          document.getElementById('loginScreen').classList.add('hidden');
          showScreen('tabHome');
        })
        .catch((error) => {
          alert("Login error: " + error.message);
        });
    }

    // DATABASE SYNC
    function loadUserData(uid) {
      const userRef = db.ref('users/' + uid);
      userRef.on('value', (snapshot) => {
        const data = snapshot.val();
        if (data) {
          userCoins = data.coins || 0;
        } else {
          // Initialize new user
          userCoins = 0;
          userRef.set({
            name: currentUser.displayName,
            email: currentUser.email,
            coins: 0,
            created_at: Date.now()
          });
        }
        updateWalletDisplay();
      });
    }

    function updateWalletDisplay() {
      document.getElementById('headerCoinBalance').innerText = userCoins;
      // 1,200 Coins = Rs 10 (1 Coin = Rs 10 / 1200 = Rs 0.00833)
      const inrValue = (userCoins * (10 / 1200)).toFixed(2);
      document.getElementById('homeInrValue').innerText = "₹" + inrValue;
    }

    // SCREEN ROUTING
    function showScreen(screenId) {
      ['tabHome', 'tabQuiz', 'tabReward', 'tabWallet'].forEach(id => {
        document.getElementById(id).classList.add('hidden');
      });
      document.getElementById(screenId).classList.remove('hidden');
    }

    // KBC ENGINE
    function startKbcQuiz() {
      currentQuizIndex = 0;
      roundCoins = 0;
      showScreen('tabQuiz');
      renderQuestion();
    }

    function renderQuestion() {
      clearInterval(quizTimer);
      secondsLeft = 15;
      document.getElementById('kbcTimer').innerText = secondsLeft;

      const q = kbcQuestions[currentQuizIndex];
      document.getElementById('kbcIndex').innerText = currentQuizIndex + 1;
      document.getElementById('kbcQuestion').innerText = q.q;

      const container = document.getElementById('kbcOptionsContainer');
      container.innerHTML = '';

      q.o.forEach((optText, index) => {
        const btn = document.createElement('button');
        btn.className = "w-full py-3 px-4 bg-slate-900/90 hover:bg-slate-800 border border-purple-900/40 rounded-xl text-left text-xs font-semibold tracking-wide transition active:scale-98 flex justify-between items-center";
        btn.innerHTML = `<span><strong class="text-amber-400 mr-2">${String.fromCharCode(65 + index)}.</strong> ${optText}</span>`;
        btn.onclick = () => submitAnswer(index);
        container.appendChild(btn);
      });

      quizTimer = setInterval(() => {
        secondsLeft--;
        document.getElementById('kbcTimer').innerText = secondsLeft;
        if (secondsLeft <= 0) {
          clearInterval(quizTimer);
          advanceQuiz();
        }
      }, 1000);
    }

    function submitAnswer(selectedIndex) {
      clearInterval(quizTimer);
      const correctAns = kbcQuestions[currentQuizIndex].a;
      if (selectedIndex === correctAns) {
        // Random reward between 8 to 12 coins
        const earned = Math.floor(Math.random() * (12 - 8 + 1)) + 8;
        roundCoins += earned;
      }
      advanceQuiz();
    }

    function advanceQuiz() {
      currentQuizIndex++;
      if (currentQuizIndex < kbcQuestions.length) {
        renderQuestion();
      } else {
        document.getElementById('roundEarnedCoins').innerText = roundCoins > 0 ? roundCoins : 10;
        showScreen('tabReward');
      }
    }

    // CPAGRIP LOCKER TRIGGER
    function triggerCpaLocker(source) {
      window.open(CPAGRIP_LOCKER_URL, '_blank');

      // Demo verification wait (Production me callback verify karein)
      setTimeout(() => {
        const coinsToAdd = (source === 'quiz_unlock') ? (roundCoins > 0 ? roundCoins : 10) : 25;
        const newBalance = userCoins + coinsToAdd;

        db.ref('users/' + currentUser.uid).update({
          coins: newBalance
        }).then(() => {
          alert(`Congratulations! ${coinsToAdd} coins credited successfully.`);
          showScreen('tabHome');
        });
      }, 3500);
    }

    // WITHDRAW / REDEEM
    function submitRedeem(requiredCoins, rupees, methodName) {
      if (userCoins < requiredCoins) {
        alert(`Insufficient Coins! You need at least ${requiredCoins} coins for ₹${rupees} withdrawal.`);
        return;
      }

      const payoutAddress = prompt(`Enter your payment detail for ${methodName} (Email or UPI ID):`);
      if (payoutAddress && payoutAddress.trim().length > 3) {
        const newBalance = userCoins - requiredCoins;

        // Record request in Firebase
        const redeemRef = db.ref('payout_requests').push();
        redeemRef.set({
          uid: currentUser.uid,
          email: currentUser.email,
          method: methodName,
          amount_inr: rupees,
          coins_deducted: requiredCoins,
          destination: payoutAddress.trim(),
          status: 'PENDING',
          timestamp: Date.now()
        }).then(() => {
          db.ref('users/' + currentUser.uid).update({ coins: newBalance });
          alert(`Redeem request for ₹${rupees} submitted successfully! Your payment status is PENDING.`);
        });
      }
    }
